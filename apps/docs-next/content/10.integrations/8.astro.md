---
title: Astro
description: "Create the request logger in Astro middleware, keep it on locals.log, and emit it when the response is done."
---

::product-facts{entry="evlog" status="stable"}
::

Astro has no evlog-specific entry point: the middleware creates a request logger from the core and emits it itself.

```typescript [src/middleware.ts]
import { defineMiddleware } from 'astro:middleware'
import { createRequestLogger, initLogger, toLoggerConfig } from 'evlog'
import config from '../evlog.config'

initLogger(toLoggerConfig(config))

export const onRequest = defineMiddleware(async ({ request, locals }, next) => {
  const log = createRequestLogger({ method: request.method, path: new URL(request.url).pathname })
  locals.log = log

  try {
    return await next()
  }
  catch (error) {
    log.error(error instanceof Error ? error : new Error(String(error)))
    throw error
  }
  finally {
    log.emit()
  }
})
```

```typescript [src/env.d.ts]
/// <reference types="astro/client" />

declare namespace App {
  interface Locals {
    log: import('evlog').RequestLogger
  }
}
```

## In an endpoint

```typescript [src/pages/api/users/[id].ts]
import type { APIRoute } from 'astro'

export const GET: APIRoute = async ({ params, locals }) => {
  const user = await db.findUser(params.id)
  locals.log.set({ user: { id: params.id, plan: user.plan } })

  return Response.json(user)
}
```

Pages rendered on the server receive `Astro.locals.log` too.
