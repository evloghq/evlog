The module registers the middleware for every route; `main.ts` starts the logger before Nest boots:

```typescript [src/app.module.ts]
import { Module } from '@nestjs/common'
import { toMiddlewareOptions } from 'evlog'
import { EvlogModule } from 'evlog/nestjs'
import config from '../evlog.config'

@Module({
  imports: [EvlogModule.forRoot(toMiddlewareOptions(config))],
})
export class AppModule {}
```

```typescript [src/main.ts]
import 'reflect-metadata'
import { NestFactory } from '@nestjs/core'
import { initLogger, toLoggerConfig } from 'evlog'
import { AppModule } from './app.module'
import config from '../evlog.config'

initLogger(toLoggerConfig(config))

const app = await NestFactory.create(AppModule)
await app.listen(3000)
```

The same module works on the Fastify platform: pass a `FastifyAdapter` to `NestFactory.create()`.

## In a controller

```typescript [src/users.controller.ts]
import { Controller, Get, Param } from '@nestjs/common'
import { useLogger } from 'evlog/nestjs'

@Controller('users')
export class UsersController {
  @Get(':id')
  async findOne(@Param('id') id: string) {
    const user = await db.findUser(id)
    useLogger().set({ user: { id, plan: user.plan } })

    return user
  }
}
```

`useLogger()` works the same way in any provider the controller calls; there is nothing to inject.
