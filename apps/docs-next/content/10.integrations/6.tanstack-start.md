TanStack Start serves through Nitro, so evlog plugs in as a Nitro module. Nitro's async context must be on, so a route can reach the request it is serving:

```typescript [nitro.config.ts]
import { defineConfig } from 'nitro'
import evlog from 'evlog/nitro'

export default defineConfig({
  experimental: { asyncContext: true },
  modules: [evlog()],
})
```

Register the error handler as server middleware on the root route, so a thrown error lands on the event:

```tsx [src/routes/__root.tsx]
import { createRootRoute } from '@tanstack/react-router'
import { createMiddleware } from '@tanstack/react-start'
import { evlogErrorHandler } from 'evlog/nitro'

export const Route = createRootRoute({
  server: {
    middleware: [createMiddleware().server(evlogErrorHandler)],
  },
})
```

## In a server route

```typescript [src/routes/api/hello.ts]
import { createFileRoute } from '@tanstack/react-router'
import { useRequest } from 'nitro/context'
import type { RequestLogger } from 'evlog'

export const Route = createFileRoute('/api/hello')({
  server: {
    handlers: {
      GET: async () => {
        const log = useRequest().context.log as RequestLogger
        log.set({ user: { id: 'user_123', plan: 'pro' } })

        return Response.json({ ok: true })
      },
    },
  },
})
```

Server functions run in the same request, so `useRequest()` reaches the same logger there.
