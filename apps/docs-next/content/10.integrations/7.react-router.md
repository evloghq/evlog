---
title: React Router
description: "Turn on React Router middleware, add evlog to the root route, and read the logger with context.get(loggerContext) in loaders and actions."
---

::product-facts{entry="evlog/react-router" status="stable"}
::

evlog runs as React Router middleware, which is behind a future flag:

```typescript [react-router.config.ts]
import type { Config } from '@react-router/dev/config'

export default {
  future: { v8_middleware: true },
} satisfies Config
```

```tsx [app/root.tsx]
import { initLogger, toLoggerConfig, toMiddlewareOptions } from 'evlog'
import { evlog } from 'evlog/react-router'
import config from '../evlog.config'

initLogger(toLoggerConfig(config))

export const middleware: Route.MiddlewareFunction[] = [
  evlog(toMiddlewareOptions(config)),
]
```

## In a loader

```typescript [app/routes/users.$id.tsx]
import { loggerContext } from 'evlog/react-router'

export async function loader({ params, context }: Route.LoaderArgs) {
  const log = context.get(loggerContext)

  const user = await db.findUser(params.id)
  log.set({ user: { id: params.id, plan: user.plan } })

  return { user }
}
```

Actions read it the same way. Underneath, `useLogger()` from `evlog/react-router` returns the logger of the current request.
